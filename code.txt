        (function() {
            const cvs = document.getElementById('bg-canvas'),
                ctx = cvs.getContext('2d');
            let W, H, pts = [];

            function resize() {
                W = cvs.width = innerWidth;
                H = cvs.height = innerHeight;
                pts = [];
                const n = Math.min(Math.floor(W * H / 14000), 110);
                for (let i = 0; i < n; i++) pts.push({
                    x: Math.random() * W,
                    y: Math.random() * H,
                    vx: (Math.random() - .5) * .3,
                    vy: (Math.random() - .5) * .3,
                    size: Math.random() * 1.5 + .5,
                    op: Math.random() * .4 + .15,
                    hue: Math.random() < .5 ? 280 : 200
                });
            }
            window.addEventListener('resize', resize);
            resize();

            function loop() {
                ctx.clearRect(0, 0, W, H);
                pts.forEach(p => {
                    p.x += p.vx;
                    p.y += p.vy;
                    if (p.x < -5 || p.x > W + 5 || p.y < -5 || p.y > H + 5) {
                        p.x = Math.random() * W;
                        p.y = Math.random() * H;
                    }
                    ctx.beginPath();
                    ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                    ctx.fillStyle = `hsla(${p.hue},75%,60%,${p.op})`;
                    ctx.fill();
                });
                for (let i = 0; i < pts.length; i++)
                    for (let j = i + 1; j < pts.length; j++) {
                        const dx = pts[i].x - pts[j].x,
                            dy = pts[i].y - pts[j].y,
                            d = Math.sqrt(dx * dx + dy * dy);
                        if (d < 90) {
                            ctx.beginPath();
                            ctx.moveTo(pts[i].x, pts[i].y);
                            ctx.lineTo(pts[j].x, pts[j].y);
                            ctx.strokeStyle = `rgba(196,77,255,${.07*(1-d/90)})`;
                            ctx.lineWidth = .5;
                            ctx.stroke();
                        }
                    }
                requestAnimationFrame(loop);
            }
            loop();
        })();

        function playFlapSound() {
            try {
                const ctx = getAudioCtx();
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.connect(gain);
                gain.connect(ctx.destination);
                osc.type = 'sine';
                osc.frequency.setValueAtTime(620, ctx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(380, ctx.currentTime + 0.08);
                gain.gain.setValueAtTime(0.25, ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.12);
                osc.start(ctx.currentTime);
                osc.stop(ctx.currentTime + 0.12);
            } catch (e) {}
        }

        function startTournament() {
            if (teams.length < 2) return toast('Need at least 2 teams!', 'error');
            const empty = teams.filter(t => t.players.length === 0);
            if (empty.length) return toast(`Add players to: ${empty.map(t=>t.name).join(', ')} ⚠️`, 'error');
            matchHistory = [];
            const n = teams.length;
            let bs = 1;
            while (bs < n) bs <<= 1;
            const sh = shuffle(teams.map(t => t.name));
            const r0 = [];
            for (let i = 0; i < bs; i += 2) r0.push({ t1: i < n ? sh[i] : 'BYE', t2: (i + 1) < n ? sh[i + 1] : 'BYE', winner: null });
            tournBracket = [r0];
            roundIdx = 0;
            matchIdx = 0;
            tournActive = true;
            toast('Tournament started! 🔥', 'success');
            updateUI();
            switchPanel('tournament');
            advanceByes();
        }

        function advanceByes() {
            if (!tournActive) return;
            const round = tournBracket[roundIdx];
            while (matchIdx < round.length) {
                const m = round[matchIdx];
                if (m.t1 === 'BYE' && m.t2 === 'BYE') { m.winner = 'BYE';
                    matchHistory.push({ round: roundIdx + 1, t1: 'BYE', t2: 'BYE', winner: 'BYE', scoreA: 0, scoreB: 0 });
                    matchIdx++; } else if (m.t1 === 'BYE') { m.winner = m.t2;
                    matchHistory.push({ round: roundIdx + 1, t1: 'BYE', t2: m.t2, winner: m.t2, scoreA: 0, scoreB: 0 });
                    matchIdx++; } else if (m.t2 === 'BYE') { m.winner = m.t1;
                    matchHistory.push({ round: roundIdx + 1, t1: m.t1, t2: 'BYE', winner: m.t1, scoreA: 0, scoreB: 0 });
                    matchIdx++; } else break;
            }
            if (matchIdx >= round.length) advanceRound();
            updateUI();
        }

        function advanceRound() {
            const round = tournBracket[roundIdx];
            const winners = round.map(m => m.winner).filter(w => w && w !== 'BYE');
            if (winners.length === 1) {
                const champ = winners[0];
                archiveCurrentTournament(champ);
                matchHistory.push({ round: 0, t1: '', t2: '', winner: champ, scoreA: 0, scoreB: 0 });
                const t = getTeam(champ);
                if (t) t.wins++;
                tournActive = false;
                updateUI();
                setTimeout(() => toast(`🏆 ${champ} IS THE CHAMPION! 🏆`, 'success'), 400);
                return;
            }
            const next = [];
            for (let i = 0; i < winners.length; i += 2)
                next.push({ t1: winners[i], t2: (i + 1 < winners.length) ? winners[i + 1] : 'BYE', winner: null });
            tournBracket.push(next);
            roundIdx++;
            matchIdx = 0;
            advanceByes();
        }

        function playNextMatch() {
            if (!tournActive) return;
            const round = tournBracket[roundIdx];
            if (matchIdx >= round.length) return;
            const m = round[matchIdx];
            if (m.t1 === 'BYE' || m.t2 === 'BYE') { advanceByes(); return; }
            openMiniGame(m.t1, m.t2, false);
        }

        class Bird {
            constructor(x, y, cfg) {
                this.x = x;
                this.y = y;
                this.r = 12;
                this.g = cfg.gravity;
                this.v = 0;
                this.j = cfg.jump;
                this.col = cfg.col;
                this.rot = 0;
                this.frame = 0;
            }
            flap() { this.v = this.j; }
            update() { this.v += this.g;
                this.y += this.v;
                this.rot = Math.min(Math.max(this.v * 2.5, -30), 30);
                this.frame++; }
            draw(ctx) {
                ctx.save();
                ctx.translate(this.x, this.y);
                ctx.rotate(this.rot * Math.PI / 180);
                ctx.fillStyle = this.col;
                ctx.beginPath();
                ctx.arc(0, 0, this.r, 0, Math.PI * 2);
                ctx.fill();
                ctx.shadowColor = this.col;
                ctx.shadowBlur = 12;
                ctx.fill();
                ctx.shadowBlur = 0;
                ctx.fillStyle = '#fff';
                ctx.beginPath();
                ctx.arc(7, -3, 3, 0, Math.PI * 2);
                ctx.fill();
                ctx.fillStyle = '#000';
                ctx.beginPath();
                ctx.arc(8, -3, 1.5, 0, Math.PI * 2);
                ctx.fill();
                ctx.fillStyle = '#e8a200';
                ctx.beginPath();
                ctx.moveTo(12, 0);
                ctx.lineTo(22, 0);
                ctx.lineTo(12, 5);
                ctx.fill();
                const wy = 4 + Math.sin(this.frame * .25) * 4;
                ctx.fillStyle = this.col + '99';
                ctx.beginPath();
                ctx.ellipse(-4, wy, 9, 5, 0, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }
            drawWithTheme(ctx, col) { this.col = col;
                this.draw(ctx); }
        }

        class FlapperGame {
            constructor(canvasId, diff, theme = 'classic', side = 'A') {
                this.cvs = document.getElementById(canvasId);
                this.ctx = this.cvs.getContext('2d');
                this.W = this.cvs.width;
                this.H = this.cvs.height;
                this.GH = 28;
                this.cfg = diff;
                this.theme = theme;
                this.side = side;
                this.bird = new Bird(65, this.H / 2, diff);
                this.pipes = [];
                this.items = [];
                this.frame = 0;
                this.score = 0;
                this.dead = false;
                this.mult = diff.mult;
                this.shieldActive = false;
                this.magnetActive = false;
                this.magnetConsumed = false;
                this.starTimer = 0;
                this.cheered = false;
                this.booed = false;
                this.flawlessPlayed = false;
                this.recording = false;
                this.frames = [];
                this.pipesSpawned = 0;
                this.nextItemAt = 15 + Math.floor(Math.random() * 10);
            }
            startRecording() { this.recording = true;
                this.frames = []; }
            stopRecording() { this.recording = false; }
            recordFrame() {
                if (!this.recording) return;
                this.frames.push({
                    birdY: this.bird.y,
                    score: this.score,
                    pipes: this.pipes.map(p => ({ x: p.x, y: p.y, w: p.w, h: p.h, passed: p.passed })),
                    items: this.items.map(it => ({ x: it.x, y: it.y, type: it.type })),
                    shield: this.shieldActive,
                    magnet: this.magnetActive,
                    starTimer: this.starTimer,
                    dead: this.dead
                });
            }
            start() {
                this.bird = new Bird(65, this.H / 2, this.cfg);
                this.pipes = [];
                this.items = [];
                this.frame = 0;
                this.score = 0;
                this.dead = false;
                this.mult = this.cfg.mult;
                this.shieldActive = false;
                this.magnetActive = false;
                this.magnetConsumed = false;
                this.starTimer = 0;
                this.cheered = false;
                this.booed = false;
                this.flawlessPlayed = false;
                this.pipesSpawned = 0;
                this.nextItemAt = 15 + Math.floor(Math.random() * 10);
            }
            flap() { if (!this.dead) this.bird.flap(); }
            update() {
                if (this.dead) return;
                this.bird.update();
                if (this.bird.y - this.bird.r <= 0 || this.bird.y + this.bird.r >= this.H - this.GH) { this.dead = true;
                    this.handleDeath(); return; }
                this.frame++;
                if (this.starTimer > 0) { this.starTimer -= 1 / 60; if (this.starTimer < 0) this.starTimer = 0; }
                if (this.frame % this.cfg.freq === 0) {
                    const py = -(Math.random() * 160 + 130);
                    this.pipes.push({ x: this.W, y: py, w: 58, h: 280 + Math.random() * 50, passed: false });
                    this.pipesSpawned++;
                    if (this.pipesSpawned >= this.nextItemAt) { this.spawnItem();
                        this.nextItemAt = this.pipesSpawned + Math.floor(Math.random() * 10) + 5; }
                }
                for (let p of this.pipes) {
                    p.x -= this.cfg.speed;
                    if (!p.passed && p.x + p.w < this.bird.x - this.bird.r) {
                        p.passed = true;
                        const added = (this.starTimer > 0) ? 2 : 1;
                        this.score += this.mult * added;
                        if (this.magnetActive && !this.magnetConsumed) { this.magnetActive = false;
                            this.magnetConsumed = true; }
                    }
                }
                this.pipes = this.pipes.filter(p => p.x + p.w > 0);
                for (let p of this.pipes) {
                    if (this.bird.x + this.bird.r > p.x && this.bird.x - this.bird.r < p.x + p.w) {
                        if (this.bird.y - this.bird.r < p.y + p.h || this.bird.y + this.bird.r > p.y + p.h + this.cfg.gap) {
                            if (this.shieldActive) { this.shieldActive = false; continue; }
                            if (this.magnetActive) continue;
                            this.dead = true;
                            this.handleDeath();
                            break;
                        }
                    }
                }
                for (let it of this.items) {
                    it.update(-this.cfg.speed);
                    if (Math.abs(it.x - this.bird.x) < 20 && Math.abs(it.y - this.bird.y) < 24) { this.collectItem(it);
                        it.collected = true; }
                }
                this.items = this.items.filter(it => !it.collected && it.x > -30);
                if (this.score >= 10 && !this.cheered) { playCommentary('cheer', this.side);
                    this.cheered = true; }
                if (this.score >= 25 && !this.flawlessPlayed) { playCommentary('flawless', this.side);
                    this.flawlessPlayed = true; }
                this.recordFrame();
            }
            handleDeath() { if (this.score < 3 && !this.booed) { playCommentary('boo', this.side);
                    this.booed = true; }
                this.recordFrame(); }
            spawnItem() {
                const types = ['shield', 'magnet', 'star'];
                const type = types[Math.floor(Math.random() * types.length)];
                const y = Math.random() * (this.H - 100) + 50;
                this.items.push(new Item(this.W + 30, y, type));
            }
            collectItem(item) {
                if (item.type === 'shield') this.shieldActive = true;
                else if (item.type === 'magnet') { this.magnetActive = true;
                    this.magnetConsumed = false; } else if (item.type === 'star') this.starTimer = 5;
                try {
                    const ctx = getAudioCtx();
                    const osc = ctx.createOscillator();
                    const gain = ctx.createGain();
                    osc.connect(gain);
                    gain.connect(ctx.destination);
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(880, ctx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(440, ctx.currentTime + 0.1);
                    gain.gain.setValueAtTime(0.12, ctx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.15);
                    osc.start(ctx.currentTime);
                    osc.stop(ctx.currentTime + 0.15);
                } catch (e) {}
            }
            draw() {
                const ctx = this.ctx;
                const th = THEMES[this.theme];
                const sky = ctx.createLinearGradient(0, 0, 0, this.H);
                sky.addColorStop(0, th.skyTop);
                sky.addColorStop(1, th.skyBottom);
                ctx.fillStyle = sky;
                ctx.fillRect(0, 0, this.W, this.H);
                for (let p of this.pipes) {
                    const pg = ctx.createLinearGradient(p.x, 0, p.x + p.w, 0);
                    pg.addColorStop(0, th.pipeDark);
                    pg.addColorStop(0.5, th.pipeColor);
                    pg.addColorStop(1, th.pipeDark);
                    ctx.fillStyle = pg;
                    ctx.fillRect(p.x, p.y, p.w, p.h);
                    ctx.fillRect(p.x, p.y + p.h + this.cfg.gap, p.w, this.H);
                    ctx.fillStyle = th.capColor;
                    ctx.fillRect(p.x - 5, p.y + p.h - 16, p.w + 10, 16);
                    ctx.fillRect(p.x - 5, p.y + p.h + this.cfg.gap, p.w + 10, 16);
                }
                ctx.fillStyle = th.ground;
                ctx.fillRect(0, this.H - this.GH, this.W, this.GH);
                ctx.fillStyle = th.groundLine;
                ctx.fillRect(0, this.H - this.GH, this.W, 3);
                for (let it of this.items) it.draw(ctx);
                this.bird.draw(ctx);
                ctx.fillStyle = 'rgba(255,255,255,0.15)';
                ctx.font = `bold 44px 'Orbitron',monospace`;
                ctx.textAlign = 'center';
                ctx.fillText(this.score, this.W / 2, this.H / 2 + 16);
                ctx.textAlign = 'left';
                if (this.shieldActive) { ctx.font = '18px sans-serif';
                    ctx.fillText('🛡️', 30, 40); }
                if (this.magnetActive) { ctx.font = '18px sans-serif';
                    ctx.fillText('🧲', 60, 40); }
                if (this.starTimer > 0) { ctx.font = '18px sans-serif';
                    ctx.fillText('⭐', 90, 40); }
            }
        }

        function aiShouldFlap(game) {
            if (game.dead) return false;
            let nextPipe = null;
            for (let p of game.pipes) { if (p.x + p.w > game.bird.x - game.bird.r) { nextPipe = p; break; } }
            if (!nextPipe) return false;
            const gapCenter = nextPipe.y + nextPipe.h + game.cfg.gap / 2;
            const birdY = game.bird.y;
            if (gapCenter > birdY + 10) return true;
            if (Math.random() < 0.2 && Math.abs(gapCenter - birdY) < 50) return true;
            return false;
        }

            if (!mgActive) return;

        function resolveMatch() {
            gA.stopRecording();
            gB.stopRecording();
            const tA = window._mgTeamA,
                tB = window._mgTeamB;
            const finalA = gA.score,
                finalB = gB.score;
            let winner;
            if (finalA > finalB) winner = tA;
            else if (finalB > finalA) winner = tB;
            else winner = Math.random() < .5 ? tA : tB;
            if (isQuickMatch) {
                matchHistory.push({
                    round: -1,
                    t1: tA,
                    t2: tB,
                    winner,
                    scoreA: finalA,
                    scoreB: finalB,
                    quickMatch: true,
                    freeMatch: (quickMatchTeamA && !getTeam(quickMatchTeamA)) || (quickMatchTeamB && !getTeam(
                        quickMatchTeamB)),
                    diffA: chosenDiffA,
                    diffB: chosenDiffB,
                    theme: gA.theme
                });
            } else {
                const round = tournBracket[roundIdx];
                round[matchIdx].winner = winner;
                matchHistory.push({
                    round: roundIdx + 1,
                    t1: tA,
                    t2: tB,
                    winner,
                    scoreA: finalA,
                    scoreB: finalB,
                    diffA: chosenDiffA,
                    diffB: chosenDiffB,
                    theme: gA.theme
                });
                matchIdx++;
            }
            showRewardSummary(tA, tB, finalA, finalB, winner);
            if (!isQuickMatch) advanceByes();
            updateUI();
            window._lastReplay = { framesA: gA.frames, framesB: gB.frames, theme: gA.theme, diffA: chosenDiffA, diffB: chosenDiffB };
        }

        }

        // ║     SECTION 17: HEX GRID BG (credit screen)    ║
        // ╚══════════════════════════════════════════════╝
        (function() {
            const cvs = document.getElementById('bg-cvs'),
                ctx = cvs.getContext('2d');
            let W, H;

            function resize() { W = cvs.width = innerWidth;
                H = cvs.height = innerHeight; }
            window.addEventListener('resize', resize);
            resize();
            const hexes = [];
            const R = 36;

            function buildHexes() { hexes.length = 0; const hx = R * Math.sqrt(3),
                    hy = R * 1.5; for (let row = -1; row < H / hy + 2; row++) { for (let col = -1; col < W / hx + 2; col++) {
                        const x = col * hx + (row % 2) * hx / 2,
                            y = row * hy;
                        hexes.push({ x, y, alpha: Math.random() * .12 + .02, pulse: Math.random() * Math.PI * 2,
                            speed: Math.random() * .008 + .002 }); } } }
            buildHexes();
            window.addEventListener('resize', buildHexes);

            function hexPath(x, y, r) { ctx.beginPath(); for (let i = 0; i < 6; i++) { const a = Math.PI / 3 * i - Math.PI / 6;
                    const px = x + r * Math.cos(a),
                        py = y + r * Math.sin(a);
                    i === 0 ? ctx.moveTo(px, py) : ctx.lineTo(px, py); }
                ctx.closePath(); }

            function loop() { ctx.clearRect(0, 0, W, H); for (const h of hexes) { h.pulse += h.speed; const a = h.alpha * (0.5 + 0.5 *
                        Math.sin(h.pulse));
                    ctx.strokeStyle = `rgba(100,60,180,${a})`;
                    ctx.lineWidth = .6;
                    hexPath(h.x, h.y, R - 2);
                    ctx.stroke(); }
                requestAnimationFrame(loop); }
            loop();
        })();
